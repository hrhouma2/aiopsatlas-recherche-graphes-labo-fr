# Module 3 — Pratique guidée : la barre de recherche du catalogue, requête par requête
# On construit LA requête qui alimente la page de recherche du catalogue de cours,
# une brique à la fois. Lecture seule sur l'index `cours` : rien n'est créé,
# rien à nettoyer.
# Rejouer : node outils/tester-requetes.mjs elasticsearch/requetes/03-pratique-la-barre-de-recherche-du-catalogue.txt

# ============================================================
# Étape 1 — La saisie « kubernetes » sur le titre
# ============================================================
GET cours/_search
{
  "query": { "match": { "titre": "kubernetes" } },
  "size": 5,
  "_source": ["id", "titre"]
}

# ============================================================
# Étape 2 — Titre (x3) et description : la saisie devient « kubernetes production »
# ============================================================
GET cours/_search
{
  "query": {
    "multi_match": {
      "query": "kubernetes production",
      "fields": ["titre^3", "description"]
    }
  },
  "size": 5,
  "_source": ["id", "titre"]
}

# ============================================================
# Étape 3 — bool : le texte dans must, les cases cochées dans filter
# ============================================================
GET cours/_search
{
  "query": {
    "bool": {
      "must": [
        { "multi_match": { "query": "kubernetes production", "fields": ["titre^3", "description"] } }
      ],
      "filter": [
        { "term":  { "categorie": "DevOps" } },
        { "terms": { "niveau": ["debutant", "avance"] } },
        { "term":  { "langue": "fr" } }
      ]
    }
  },
  "size": 5,
  "_source": ["id", "titre", "niveau", "prix"]
}

# ============================================================
# Étape 4 — Budget : prix <= 50 $ ou gratuit
# ============================================================
GET cours/_search
{
  "query": {
    "bool": {
      "must": [
        { "multi_match": { "query": "kubernetes production", "fields": ["titre^3", "description"] } }
      ],
      "filter": [
        { "term":  { "categorie": "DevOps" } },
        { "terms": { "niveau": ["debutant", "avance"] } },
        { "term":  { "langue": "fr" } },
        { "bool": {
            "should": [
              { "range": { "prix": { "lte": 50 } } },
              { "term":  { "gratuit": true } }
            ],
            "minimum_should_match": 1
        } }
      ]
    }
  },
  "size": 10,
  "_source": ["id", "titre", "prix", "gratuit"]
}

        { "terms": { "niveau": ["debutant", "avance"] } },

            "should": [
              { "range": { "prix": { "lte": 50 } } },

            "minimum_should_match": 1
        } }
      ]
    }
  },

      "filter": [
        { "term":  { "categorie": "DevOps" } },

        } }
      ]
    }

# ============================================================
# Étape 6 — Bonus d'expression : match_phrase + slop dans should
# ============================================================

# 6a. Diagnostic : l'expression exacte, sans slop, ne trouve rien
GET cours/_search
{
  "query": { "match_phrase": { "titre": "kubernetes production" } },
  "size": 3,
  "_source": ["titre"]
}

# 6b. Diagnostic : avec slop 5, les titres où les deux mots sont proches
GET cours/_search
{
  "query": { "match_phrase": { "titre": { "query": "kubernetes production", "slop": 5 } } },
  "size": 3,
  "_source": ["titre"]
}

# 6c. La brique dans la requête : should ne filtre pas, il récompense
GET cours/_search
{
  "query": {
    "bool": {
      "must": [
        { "multi_match": {
            "query": "kubernetes production",
            "fields": ["titre^3", "description"],
            "fuzziness": "AUTO"
        } }
      ],
      "filter": [
        { "term":  { "categorie": "DevOps" } },
        { "terms": { "niveau": ["debutant", "avance"] } },
        { "term":  { "langue": "fr" } },
        { "bool": {
            "should": [
              { "range": { "prix": { "lte": 50 } } },
              { "term":  { "gratuit": true } }
            ],
            "minimum_should_match": 1
        } }
      ],
      "should": [
        { "match_phrase": { "titre": { "query": "kubernetes production", "slop": 5, "boost": 2 } } }
      ]
    }
  },
  "size": 3,
  "_source": ["id", "titre"]
}

# ============================================================
# Si ça coince : les erreurs classiques de cette pratique
# ============================================================

# S1. filter posé dans multi_match au lieu de bool
# attendu: 400
GET cours/_search
{
  "query": {
    "multi_match": {
      "query": "kubernetes production",
      "fields": ["titre^3", "description"],
      "filter": [ { "term": { "categorie": "DevOps" } } ]
    }
  }
}

# S2. terms avec une valeur seule au lieu d'un tableau
# attendu: 400
GET cours/_search
{
  "query": { "bool": { "filter": [ { "terms": { "niveau": "debutant" } } ] } }
}

# S3. search_after avec un tableau de 2 valeurs pour un sort à 3 critères
# attendu: 400
GET cours/_search
{
  "query": { "match": { "titre": "kubernetes" } },
  "sort": [ { "note_moyenne": "desc" }, "_score", { "id": "asc" } ],
  "search_after": [4.7, "C0084"],
  "size": 3
}

# S4. fuzziness dans match_phrase
# attendu: 400
GET cours/_search
{
  "query": { "match_phrase": { "titre": { "query": "kubernetes production", "slop": 5, "fuzziness": "AUTO" } } }
}

# S5. highlight posé dans le bool au lieu d'être à côté de query
# attendu: 400
GET cours/_search
{
  "query": {
    "bool": {
      "must": [ { "match": { "titre": "kubernetes" } } ],
      "highlight": { "fields": { "titre": {} } }
    }
  }
}

# S6. Pas d'erreur mais 0 résultat : la casse du keyword
GET cours/_count
{
  "query": { "bool": { "filter": [ { "term": { "categorie": "devops" } } ] } }
}
