# 3. Recherche « docker » + filtres (niveau débutant, prix <= 50) + 4 facettes.
#    Les filtres sont dans bool.filter : les compteurs des facettes portent sur
#    le résultat FILTRÉ (ce que l'utilisateur voit à l'écran).
GET cours/_search
{
  "size": 5,
  "_source": ["id", "titre", "categorie", "niveau", "prix", "gratuit"],
  "query": {
    "bool": {
      "must": [
        {
          "multi_match": {
            "query": "docker",
            "fields": ["titre^3", "description"],
            "fuzziness": "AUTO"
          }
        }
      ],
      "filter": [
        { "term": { "niveau": "debutant" } },
        { "range": { "prix": { "lte": 50 } } }
      ]
    }
  },
  "aggs": {
    "categories": { "terms": { "field": "categorie" } },
    "niveaux": { "terms": { "field": "niveau" } },
    "gratuit": { "terms": { "field": "gratuit" } },
    "tranches_de_prix": {
      "range": {
        "field": "prix",
        "ranges": [
          { "key": "gratuit", "to": 0.01 },
          { "key": "moins de 50", "from": 0.01, "to": 50 },
          { "key": "de 50 à 100", "from": 50, "to": 100 },
          { "key": "100 et plus", "from": 100 }
        ]
      }
    }
  }
}

# 4. Variante « vraies facettes » : les filtres passent dans post_filter.
#    Les hits sont filtrés, mais les compteurs restent ceux de la recherche
#    complète « docker » (39 cours) : l'utilisateur voit les autres options.
GET cours/_search
{
  "size": 5,
  "_source": ["id", "titre", "categorie", "niveau", "prix", "gratuit"],
  "query": {
    "multi_match": {
      "query": "docker",
      "fields": ["titre^3", "description"],
      "fuzziness": "AUTO"
    }
  },
  "post_filter": {
    "bool": {
      "filter": [
        { "term": { "niveau": "debutant" } },
        { "range": { "prix": { "lte": 50 } } }
      ]
    }
  },
  "aggs": {
    "categories": { "terms": { "field": "categorie" } },
    "niveaux": { "terms": { "field": "niveau" } },
    "gratuit": { "terms": { "field": "gratuit" } },
    "tranches_de_prix": {
      "range": {
        "field": "prix",
        "ranges": [
          { "key": "gratuit", "to": 0.01 },
          { "key": "moins de 50", "from": 0.01, "to": 50 },
          { "key": "de 50 à 100", "from": 50, "to": 100 },
          { "key": "100 et plus", "from": 100 }
        ]
      }
    }
  }
}

# 6. Le tri NAÏF par note_moyenne : un cours avec 27 avis passe devant un cours
#    avec 260 avis. C'est le piège à éviter.
GET cours/_search
{
  "size": 5,
  "_source": ["id", "titre", "note_moyenne", "nb_avis"],
  "query": {
    "multi_match": {
      "query": "docker",
      "fields": ["titre^3", "description"],
      "fuzziness": "AUTO"
    }
  },
  "sort": [ { "note_moyenne": "desc" } ]
}

# 7. Tri « mieux notés » : note pondérée (moyenne bayésienne).
#    score = (note * nb_avis + C * m) / (nb_avis + m)
#    C = 4.1 (note moyenne du catalogue), m = 50 (nombre d'avis « de confiance »).
#    Un cours sans note (note_moyenne absente) compte comme 0.
GET cours/_search
{
  "size": 5,
  "_source": ["id", "titre", "note_moyenne", "nb_avis"],
  "query": {
    "multi_match": {
      "query": "docker",
      "fields": ["titre^3", "description"],
      "fuzziness": "AUTO"
    }
  },
  "sort": [
    {
      "_script": {
        "type": "number",
        "order": "desc",
        "script": {
          "lang": "painless",
          "source": "double n = doc['nb_avis'].value; double note = doc['note_moyenne'].size() == 0 ? 0 : doc['note_moyenne'].value; return (note * n + params.C * params.m) / (n + params.m);",
          "params": { "C": 4.1, "m": 50 }
        }
      }
    }
  ]
}

# 8. « kub » : match_phrase_prefix travaille sur les mots analysés (minuscules,
#    sans accents, racinisés) : « Kubernetes », « kubernetes », « KUBERNETES » sont trouvés
GET cours/_search
{
  "size": 5,
  "_source": ["titre"],
  "query": {
    "match_phrase_prefix": { "titre": "kub" }
  }
}

# 9. Plusieurs mots, le dernier incomplet : match_bool_prefix
GET cours/_search
{
  "size": 5,
  "_source": ["titre"],
  "query": {
    "match_bool_prefix": { "titre": "les bases de kub" }
  }
}

# 10. Le piège : prefix sur le keyword titre.brut est sensible à la casse et
#     ne regarde que le DÉBUT du titre → « Les bases de Kubernetes » est raté
GET cours/_search
{
  "size": 5,
  "_source": ["titre"],
  "query": {
    "prefix": { "titre.brut": "Kub" }
  }
}

# 14. Taux d'erreur 5xx par jour : date_histogram + filter statut >= 500
#     + bucket_script pour calculer le pourcentage dans chaque jour
GET acces/_search
{
  "size": 0,
  "aggs": {
    "par_jour": {
      "date_histogram": { "field": "@timestamp", "calendar_interval": "day" },
      "aggs": {
        "erreurs_5xx": {
          "filter": { "range": { "statut": { "gte": 500 } } }
        },
        "taux_5xx": {
          "bucket_script": {
            "buckets_path": { "err": "erreurs_5xx._count", "total": "_count" },
            "script": "100.0 * params.err / params.total"
          }
        }
      }
    }
  }
}

# 15. Bonus : les mêmes chiffres en ES|QL (vu au module 3, leçon 5)
POST _query?format=txt
{
  "query": "FROM acces | WHERE cours_id IS NOT NULL | STATS consultations = COUNT(*) BY cours_id | SORT consultations DESC, cours_id | LIMIT 10"
}

POST _query?format=txt
{
  "query": "FROM acces | EVAL jour = DATE_TRUNC(1 day, @timestamp), erreur = CASE(statut >= 500, 1, 0) | STATS total = COUNT(*), erreurs_5xx = SUM(erreur) BY jour | EVAL taux_5xx = ROUND(100.0 * erreurs_5xx / total, 2) | SORT jour | LIMIT 40"
}
