# Module 5 — Pratique guidée : migrer la recherche du catalogue vers OpenSearch (côté OPENSEARCH, port 9201)
#   node outils/tester-requetes.mjs elasticsearch/requetes/05-pratique-migration-opensearch.txt --url http://localhost:9201
# Le fichier jumeau 05-pratique-migration-elasticsearch.txt rejoue la même pratique sur 9200.
# Rejouable : l'index pratique-migration est (re)créé en tête et supprimé à la fin.

# 0. Repartir propre (ne fait rien si l'index n'existe pas)
DELETE pratique-migration?ignore_unavailable=true

# Étape 1 — Qui répond ? (distribution: opensearch + tagline « The OpenSearch Project »)
GET /

# Étape 2 — Créer l'index avec le même mapping que sur Elasticsearch
PUT pratique-migration
{
  "settings": {
    "number_of_shards": 1,
    "number_of_replicas": 0,
    "analysis": { "analyzer": { "francais": { "type": "french" } } }
  },
  "mappings": {
    "properties": {
      "titre": { "type": "text", "analyzer": "francais" },
      "categorie": { "type": "keyword" },
      "prix": { "type": "float" }
    }
  }
}

# Étape 3 — La recherche du catalogue : bool must + filter
GET pratique-migration/_search
{
  "query": {
    "bool": {
      "must": [ { "match": { "titre": "déployer conteneurs" } } ],
      "filter": [ { "term": { "categorie": "DevOps" } }, { "range": { "prix": { "lt": 60 } } } ]
    }
  },
  "_source": ["titre", "prix"]
}

# Étape 4 — Faute de frappe : fuzzy
GET pratique-migration/_search
{
  "query": { "match": { "titre": { "query": "kubernetess", "fuzziness": "AUTO" } } },
  "_source": ["titre"]
}

# Étape 4 — La facette « catégorie » : agrégation terms
GET pratique-migration/_search
{
  "size": 0,
  "aggs": { "par_categorie": { "terms": { "field": "categorie", "size": 10 } } }
}

# Étape 5 — Ce qui n'existe pas côté OpenSearch : ES|QL → 405 (« _query » pris pour un nom d'index)
# attendu: 405
POST _query?format=txt
{
  "query": "FROM pratique-migration | STATS n = COUNT(*) BY categorie | SORT n DESC, categorie ASC"
}

# Étape 6 — La facette réécrite en SQL (plugin opensearch-sql)
POST _plugins/_sql
{
  "query": "SELECT categorie, COUNT(*) AS n FROM pratique-migration GROUP BY categorie ORDER BY n DESC, categorie ASC"
}

# Étape 6 — La même facette en PPL
POST _plugins/_ppl
{
  "query": "source=pratique-migration | stats count() as n by categorie | sort - n, categorie"
}

# Étape 9 — Vérification finale : 8 documents et la liste des plugins
GET pratique-migration/_count

GET _cat/plugins?v&h=component,version&s=component

# Nettoyage
DELETE pratique-migration
