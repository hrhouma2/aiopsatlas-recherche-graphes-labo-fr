# Module 2 — Pratique guidée : Construire l'index pratique-bibliotheque de A à Z
# Rejouable : crée les index pratique-bibliotheque et pratique-bibliotheque-v2, les supprime à la fin.
# Ne touche pas à cours/avis/acces.
# Note : DELETE pratique-* est refusé par Elasticsearch (action.destructive_requires_name = true
# par défaut) → on nomme les index explicitement.

# 0. Nettoyage au cas où un essai précédent aurait laissé les index (404 ignoré)
DELETE pratique-bibliotheque,pratique-bibliotheque-v2?ignore_unavailable=true

# ---------------------------------------------------------------------------
# Étape 1 — Créer l'index avec son mapping (et l'alias de lecture)
# ---------------------------------------------------------------------------
PUT pratique-bibliotheque
{
  "settings": {
    "number_of_shards": 1,
    "number_of_replicas": 0,
    "analysis": { "analyzer": { "francais": { "type": "french" } } }
  },
  "aliases": { "pratique-bibliotheque-lecture": {} },
  "mappings": {
    "properties": {
      "id":       { "type": "keyword" },
      "titre":    { "type": "text", "analyzer": "francais", "fields": { "brut": { "type": "keyword" } } },
      "resume":   { "type": "text", "analyzer": "francais" },
      "prix":     { "type": "float" },
      "pages":    { "type": "integer" },
      "parution": { "type": "date" },
      "themes":   { "type": "keyword" },
      "auteur": {
        "properties": {
          "nom":   { "type": "text", "fields": { "brut": { "type": "keyword" } } },
          "ville": { "type": "keyword" }
        }
      }
    }
  }
}

# ---------------------------------------------------------------------------
# Étape 7 — Provoquer une erreur de mapping : un champ deviné avec le mauvais type
# ---------------------------------------------------------------------------
# Un collègue ajoute une note des lecteurs sur un livre : 4 → Elasticsearch devine long
POST pratique-bibliotheque/_update/L003?refresh=true
{
  "doc": { "note_lecteurs": 4 }
}

# Puis 4.7 sur un autre livre : accepté, mais indexé comme 4
POST pratique-bibliotheque/_update/L007?refresh=true
{
  "doc": { "note_lecteurs": 4.7 }
}

GET pratique-bibliotheque/_mapping/field/note_lecteurs

# Les livres notés 4,5 et plus : aucun, alors que L007 a 4,7
GET pratique-bibliotheque/_search
{
  "query": { "range": { "note_lecteurs": { "gte": 4.5 } } },
  "_source": ["id", "note_lecteurs"]
}

# Tenter de corriger le type : refusé
# attendu: 400
PUT pratique-bibliotheque/_mapping
{
  "properties": { "note_lecteurs": { "type": "float" } }
}

    "properties": {
      "id":       { "type": "keyword" },
      "titre":    { "type": "text", "analyzer": "francais", "fields": { "brut": { "type": "keyword" } } },

      "prix":     { "type": "float" },
      "pages":    { "type": "integer" },
      "parution": { "type": "date" },
      "themes":   { "type": "keyword" },
      "note_lecteurs": { "type": "float" },
      "auteur": {
        "properties": {
          "nom":   { "type": "text", "fields": { "brut": { "type": "keyword" } } },
          "ville": { "type": "keyword" }

      }
    }
  }
}
